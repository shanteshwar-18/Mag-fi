// to avoid OOM on low-RAM devices. Log bitmap dimensions after load.

// ── setScaleFromTwoPoints() ───────────────────────────────────────────
// Computes pixel distance: pixDist = sqrt((px2-px1)^2 + (py2-py1)^2)
// Computes real distance: realDist = sqrt((rx2-rx1)^2 + (ry2-ry1)^2)
// scalePxPerMeter = pixDist / realDist
// Example: Door pixel=(50,200), Corridor end pixel=(50,600), real dist=8m
//   → scalePxPerMeter = 400/8 = 50

// ── metersToPixels(x, y) ─────────────────────────────────────────────
// Converts PDR meter coordinates to canvas pixel coordinates:
//   px = originPixelX + (xMeters * scalePxPerMeter)
//   py = originPixelY - (yMeters * scalePxPerMeter)  // Y flipped: map Y-down
// Returns Pair<Float, Float>(px, py)

class FingerprintMatcher {
    companion object {
        const val K = 3
        const val SNAP_THRESHOLD_UT = 8.0f   // µT — snap only if best SSD < this
    }

    // ── matchPosition() — THE FULL 3-LAYER PIPELINE ──────────────────
    // Returns: Position(x,y) if a valid snap should occur, null otherwise.
    fun matchPosition(
        allFingerprints: List<Fingerprint>,
        liveBx: Float, liveBy: Float, liveBz: Float,
        strongestRouter: Pair<String, Int>?   // from WifiScanner
    ): Pair<Float, Float>? { ... }
}

// ── LAYER 1: Wi-Fi Coarse Filter ─────────────────────────────────────
// Input: allFingerprints (all rows from DB), strongestRouter SSID
// Step 1: If strongestRouter is null → skip Wi-Fi filter, use ALL fingerprints.
// Step 2: For each fingerprint, parse its wifiPayload JSON.
//         Retain only fingerprints whose JSON CONTAINS the strongestRouter SSID.
//         This restricts the search to the same Wi-Fi macro-zone.
// Step 3: If filtered list is empty (no match for this router) → use ALL fingerprints.
//         (fail-safe: PDR-only mode)
// Output: candidates: List<Fingerprint>

// ── LAYER 2: Magnetic SSD Computation ────────────────────────────────
// For each candidate fingerprint, compute Sum of Squared Differences:
//   SSD = sqrt(
//       (liveBx - fp.magX)^2 +
//       (liveBy - fp.magY)^2 +
//       (liveBz - fp.magZ)^2
//   )
// Attach SSD to each candidate.
// Sort candidates ascending by SSD.

// ── LAYER 3: KNN Position Estimate ────────────────────────────────────
// Take top K=3 candidates (lowest SSD).
// Snap condition check: if candidates[0].SSD >= SNAP_THRESHOLD_UT → return null
//   (magnetic mismatch — don't snap, let PDR continue).
// Compute mean position:
//   avgX = (candidates[0].posX + candidates[1].posX + candidates[2].posX) / 3
//   avgY = (candidates[0].posY + candidates[1].posY + candidates[2].posY) / 3
// Return Pair(avgX, avgY)

// ── LOOP CLOSURE ──────────────────────────────────────────────────────
// If the user revisits a mapped area, the KNN match will naturally return
// a high-confidence (low SSD) result for that area, re-aligning drift.
// No extra code needed — this is an emergent property of the snap mechanism.

data class GraphNode(val id: String, val x: Float, val y: Float)

class RoutingEngine {
    private val nodes = mutableMapOf<String, GraphNode>()
    private val adjacency = mutableMapOf<String, MutableList<Pair<String,Float>>>()

    fun addNode(id: String, x: Float, y: Float) { ... }
    fun addEdge(fromId: String, toId: String) { ... }   // auto-weight = Euclidean dist
    fun findPath(startId: String, endId: String): List<GraphNode>? { ... }
    fun getNode(id: String): GraphNode? = nodes[id]
    fun getAllNodeIds(): List<String> = nodes.keys.toList()
}

// ── addEdge() ─────────────────────────────────────────────────────────
// Computes Euclidean distance between the two nodes as edge weight.
// Adds BOTH directions (undirected graph):
//   adjacency[fromId].add(Pair(toId, dist))
//   adjacency[toId].add(Pair(fromId, dist))

// ── findPath() — Dijkstra ─────────────────────────────────────────────
// Standard Dijkstra with PriorityQueue<Pair<Float,String>>:
// dist = mutableMapOf<String,Float>() initialized to Float.MAX_VALUE
// prev = mutableMapOf<String,String?>()
// dist[startId] = 0f
// PriorityQueue comparator: ascending by distance
// While queue not empty:
//   Pop (cost, nodeId) with lowest cost
//   If nodeId == endId: reconstruct path using prev map, return
//   For each neighbor: relax edge
// Returns null if endId is unreachable.
// Reconstructed path: List<GraphNode> from start to end in order.

fun setDestination(name: String) {
    destinationName.postValue(name)
    // Find closest node whose id or display name contains 'name' (case-insensitive).

        .firstOrNull { it.contains(name.trim(), ignoreCase=true) }

    }

                                    pdrTracker.getPosition().y)

    } else {
        currentRoute.postValue(path)
    }
}

// ── findNearestNode(x, y): String ────────────────────────────────────
// Returns the node ID with the smallest Euclidean distance to (x,y).
// Used to snap the start of the route to the user's current position.
private fun findNearestNode(x: Float, y: Float): String {
    return routingEngine.getAllNodeIds().minByOrNull { id ->
        val n = routingEngine.getNode(id)!!
        val dx = n.x - x; val dy = n.y - y
        dx*dx + dy*dy
    } ?: routingEngine.getAllNodeIds().first()
}
